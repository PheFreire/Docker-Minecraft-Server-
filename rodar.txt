# ============================================================
#  SETUP MINECRAFT FORGE 1.20.1 + MODS
# ============================================================

# PASSO 1 — Baixar o instalador do Forge 1.20.1
curl -L -o ~/Downloads/forge-1.20.1-installer.jar \
  "https://maven.minecraftforge.net/net/minecraftforge/forge/1.20.1-47.2.0/forge-1.20.1-47.2.0-installer.jar"

# PASSO 2 — Instalar o Forge
java -jar ~/Downloads/forge-1.20.1-installer.jar

# PASSO 3 — Criar a pasta de mods
mkdir -p ~/Library/Application\ Support/minecraft/mods
