# ============================================================
#  SETUP MINECRAFT FORGE 1.20.1 + MODS (WINDOWS)
# ============================================================
# Rode estes comandos no Prompt de Comando (cmd.exe).
# Se for usar o PowerShell, troque "curl" por "curl.exe" em cada
# linha (no PowerShell "curl" é um alias do Invoke-WebRequest e
# não entende as flags -L/-o do jeito certo).

# PASSO 1 — Baixar o instalador do Forge 1.20.1
curl.exe -L -o "%USERPROFILE%\Downloads\forge-1.20.1-installer.jar" "https://maven.minecraftforge.net/net/minecraftforge/forge/1.20.1-47.2.0/forge-1.20.1-47.2.0-installer.jar"

# PASSO 2 — Instalar o Forge
java -jar "%USERPROFILE%\Downloads\forge-1.20.1-installer.jar"

# PASSO 3 — Criar a pasta de mods
mkdir "%APPDATA%\.minecraft\mods"

# PASSO 4 — Baixar todos os mods do servidor
cd /d "%APPDATA%\.minecraft\mods"

curl.exe -L -o Chunky-1.3.146.jar "https://cdn.modrinth.com/data/fALzjamp/versions/4FTDk9wv/Chunky-1.3.146.jar"

curl.exe -L -o citadel-2.6.3-1.20.1.jar "https://cdn.modrinth.com/data/jJfV67b1/versions/lTAAe4sZ/citadel-2.6.3-1.20.1.jar"
